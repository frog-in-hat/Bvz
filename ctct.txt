CREATE TABLE Оценки (
    id integer AUTOINCREMENT PRIMARY KEY,
    Доклад_id INT NOT NULL UNIQUE ,
    Участник_id  INT NOT NULL UNIQUE ,
    score INT NOT NULL CHECK (score BETWEEN 1 AND 10),
    FOREIGN KEY (Доклад_id) REFERENCES Доклады(id),
    FOREIGN KEY (Участник_id) REFERENCES Участники(id)
);

CREATE TABLE Модераторы (
    attendance_id   SERIAL PRIMARY KEY,
    participant_id  INT NOT NULL,
    section_id      INT NOT NULL,
    attended_at     TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE (participant_id, section_id),
    FOREIGN KEY (participant_id) REFERENCES Participant(participant_id) ON DELETE CASCADE,
    FOREIGN KEY (section_id) REFERENCES Section(section_id) ON DELETE CASCADE
);
